=== TS-3.2 Media service

Implements <<FR-1.3>> and <<FR-1.4>>.

[source,typescript]
----
class MediaService {
  // Camera
  async captureImage(): Promise<Blob>;
  async selectImages(): Promise<Blob[]>;
  async compressImage(blob: Blob, maxSizeMB: number): Promise<Blob>;

  // Audio
  async startRecording(): Promise<MediaRecorder>;
  async stopRecording(recorder: MediaRecorder): Promise<Blob>;
}
